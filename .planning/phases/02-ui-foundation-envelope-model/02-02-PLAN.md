// Envelope activity container
// Each envelope holds one activity (Would You Rather, Letter, Trivia, etc.)
model Envelope {
  id          String   @id @default(uuid())
  title       String   // Display title (e.g., "Would You Rather #1")
  type        String   // Activity type: 'would-you-rather' | 'letter' | 'trivia' | 'name-game' | 'gender-reveal'
  status      String   @default("sealed") // 'sealed' | 'opened' | 'completed'
  order       Int      // Display order in pile
  createdAt   DateTime @default(now()) @map("created_at")
  updatedAt   DateTime @updatedAt @map("updated_at")

  @@map("envelopes")
}
```

import { z } from 'zod';

/**
 * Envelope types for Before We Were Three
 * Envelopes are the core UI metaphor - each contains one activity
 */

/**
 * Valid envelope statuses
 * - sealed: Not yet opened by users
 * - opened: Currently being worked on
 * - completed: Activity finished
 */
export type EnvelopeStatus = 'sealed' | 'opened' | 'completed';

/**
 * Valid envelope/activity types
 */
export type EnvelopeType =
  | 'would-you-rather'
  | 'letter'
  | 'trivia'
  | 'name-game'
  | 'gender-reveal';

/**
 * Envelope entity matching database model
 */
export interface Envelope {
  id: string;
  title: string;
  type: EnvelopeType;
  status: EnvelopeStatus;
  order: number;
  createdAt: string; // ISO date string for API transport
  updatedAt: string;
}

/**
 * Schema for creating an envelope (admin only)
 */
export const createEnvelopeSchema = z.object({
  title: z.string().min(1, 'Title is required').max(100, 'Title too long'),
  type: z.enum(['would-you-rather', 'letter', 'trivia', 'name-game', 'gender-reveal']),
  order: z.number().int().min(0, 'Order must be non-negative'),
});

export type CreateEnvelopeRequest = z.infer<typeof createEnvelopeSchema>;

/**
 * Schema for updating an envelope (admin only)
 */
export const updateEnvelopeSchema = z.object({
  title: z.string().min(1).max(100).optional(),
  type: z.enum(['would-you-rather', 'letter', 'trivia', 'name-game', 'gender-reveal']).optional(),
  status: z.enum(['sealed', 'opened', 'completed']).optional(),
  order: z.number().int().min(0).optional(),
});

export type UpdateEnvelopeRequest = z.infer<typeof updateEnvelopeSchema>;

/**
 * Response types for envelope API
 */
export interface EnvelopeListResponse {
  envelopes: Envelope[];
}

export interface EnvelopeResponse {
  envelope: Envelope;
}
```

```typescript
import { PrismaClient, Envelope as PrismaEnvelope } from '@prisma/client';
import type { Envelope, CreateEnvelopeRequest, UpdateEnvelopeRequest } from 'shared';

const prisma = new PrismaClient();

/**
 * Transform Prisma Envelope to API Envelope
 * Converts Date objects to ISO strings for transport
 */
function toApiEnvelope(envelope: PrismaEnvelope): Envelope {
  return {
    ...envelope,
    type: envelope.type as Envelope['type'],
    status: envelope.status as Envelope['status'],
    createdAt: envelope.createdAt.toISOString(),
    updatedAt: envelope.updatedAt.toISOString(),
  };
}

/**
 * Get all envelopes ordered by order field
 */
export async function getAllEnvelopes(): Promise<Envelope[]> {
  const envelopes = await prisma.envelope.findMany({
    orderBy: { order: 'asc' },
  });
  return envelopes.map(toApiEnvelope);
}

/**
 * Get single envelope by ID
 */
export async function getEnvelopeById(id: string): Promise<Envelope | null> {
  const envelope = await prisma.envelope.findUnique({
    where: { id },
  });
  return envelope ? toApiEnvelope(envelope) : null;
}

/**
 * Create new envelope (admin only)
 */
export async function createEnvelope(data: CreateEnvelopeRequest): Promise<Envelope> {
  const envelope = await prisma.envelope.create({
    data: {
      title: data.title,
      type: data.type,
      order: data.order,
      status: 'sealed', // New envelopes always start sealed
    },
  });
  return toApiEnvelope(envelope);
}

/**
 * Update envelope (admin only)
 */
export async function updateEnvelope(
  id: string,
  data: UpdateEnvelopeRequest
): Promise<Envelope | null> {
  try {
    const envelope = await prisma.envelope.update({
      where: { id },
      data,
    });
    return toApiEnvelope(envelope);
  } catch {
    // Prisma throws if record not found
    return null;
  }
}

/**
 * Delete envelope (admin only)
 */
export async function deleteEnvelope(id: string): Promise<boolean> {
  try {
    await prisma.envelope.delete({
      where: { id },
    });
    return true;
  } catch {
    return false;
  }
}

/**
 * Update envelope status (for user interactions)
 */
export async function updateEnvelopeStatus(
  id: string,
  status: Envelope['status']
): Promise<Envelope | null> {
  return updateEnvelope(id, { status });
}
```

3. Create server/src/routes/envelopes.ts:

```typescript
import { Router, Request, Response } from 'express';
import {
  successResponse,
  errorResponse,
  createEnvelopeSchema,
  updateEnvelopeSchema,
} from 'shared';
import {
  getAllEnvelopes,
  getEnvelopeById,
  createEnvelope,
  updateEnvelope,
  deleteEnvelope,
} from '../db/queries/envelopes.js';
import { requireAuth, requireAdmin } from '../middleware/auth.js';

export const envelopesRouter = Router();

/**
 * GET /api/envelopes
 * List all envelopes (requires authentication)
 */
envelopesRouter.get('/', requireAuth, async (_req: Request, res: Response) => {
  try {
    const envelopes = await getAllEnvelopes();
    res.json(successResponse({ envelopes }));
  } catch (error) {
    console.error('Failed to fetch envelopes:', error);
    res.status(500).json(errorResponse('INTERNAL_ERROR', 'Failed to fetch envelopes'));
  }
});

/**
 * GET /api/envelopes/:id
 * Get single envelope by ID (requires authentication)
 */
envelopesRouter.get('/:id', requireAuth, async (req: Request, res: Response) => {
  try {
    const envelope = await getEnvelopeById(req.params.id);
    if (!envelope) {
      res.status(404).json(errorResponse('ENVELOPE_NOT_FOUND', 'Envelope not found'));
      return;
    }
    res.json(successResponse({ envelope }));
  } catch (error) {
    console.error('Failed to fetch envelope:', error);
    res.status(500).json(errorResponse('INTERNAL_ERROR', 'Failed to fetch envelope'));
  }
});

/**
 * POST /api/envelopes
 * Create new envelope (admin only)
 */
envelopesRouter.post('/', requireAuth, requireAdmin, async (req: Request, res: Response) => {
  try {
    const parsed = createEnvelopeSchema.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json(
        errorResponse('VALIDATION_ERROR', 'Invalid envelope data', {
          issues: parsed.error.issues,
        })
      );
      return;
    }

    const envelope = await createEnvelope(parsed.data);
    res.status(201).json(successResponse({ envelope }));
  } catch (error) {
    console.error('Failed to create envelope:', error);
    res.status(500).json(errorResponse('INTERNAL_ERROR', 'Failed to create envelope'));
  }
});

/**
 * PATCH /api/envelopes/:id
 * Update envelope (admin only)
 */
envelopesRouter.patch('/:id', requireAuth, requireAdmin, async (req: Request, res: Response) => {
  try {
    const parsed = updateEnvelopeSchema.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json(
        errorResponse('VALIDATION_ERROR', 'Invalid envelope data', {
          issues: parsed.error.issues,
        })
      );
      return;
    }

    const envelope = await updateEnvelope(req.params.id, parsed.data);
    if (!envelope) {
      res.status(404).json(errorResponse('ENVELOPE_NOT_FOUND', 'Envelope not found'));
      return;
    }

    res.json(successResponse({ envelope }));
  } catch (error) {
    console.error('Failed to update envelope:', error);
    res.status(500).json(errorResponse('INTERNAL_ERROR', 'Failed to update envelope'));
  }
});

/**
 * DELETE /api/envelopes/:id
 * Delete envelope (admin only)
 */
envelopesRouter.delete('/:id', requireAuth, requireAdmin, async (req: Request, res: Response) => {
  try {
    const deleted = await deleteEnvelope(req.params.id);
    if (!deleted) {
      res.status(404).json(errorResponse('ENVELOPE_NOT_FOUND', 'Envelope not found'));
      return;
    }

    res.status(204).send();
  } catch (error) {
    console.error('Failed to delete envelope:', error);
    res.status(500).json(errorResponse('INTERNAL_ERROR', 'Failed to delete envelope'));
  }
});
```
