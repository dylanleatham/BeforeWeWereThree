```typescript
import type { WYRPromptResponse, WYRVoteRequest, WYRVoteResponse, WYRPrompt, CreateWyrPromptRequest, UpdateWyrPromptRequest } from 'shared';

/**
 * Get WYR prompt state for an envelope
 */
export async function getWyrPrompt(envelopeId: string): Promise<WYRPromptResponse> {
  const response = await apiFetch<WYRPromptResponse>(`/wyr/${envelopeId}`);
  if (!response.success) {
    throw new Error(response.error?.message || 'Failed to fetch WYR prompt');
  }
  return response.data;
}

/**
 * Submit vote for a WYR prompt
 */
export async function submitWyrVote(promptId: string, choice: 'option_a' | 'option_b'): Promise<WYRVoteResponse> {
  const response = await apiFetch<WYRVoteResponse>(`/wyr/${promptId}/vote`, {
    method: 'POST',
    body: JSON.stringify({ choice }),
  });
  if (!response.success) {
    throw new Error(response.error?.message || 'Failed to submit vote');
  }
  return response.data;
}

/**
 * Create WYR prompt (admin only)
 */
export async function createWyrPrompt(data: { envelopeId: string; optionA: string; optionB: string }): Promise<WYRPrompt> {
  const response = await apiFetch<{ prompt: WYRPrompt }>('/wyr', {
    method: 'POST',
    body: JSON.stringify(data),
  });
  if (!response.success) {
    throw new Error(response.error?.message || 'Failed to create WYR prompt');
  }
  return response.data.prompt;
}

/**
 * Update WYR prompt (admin only)
 */
export async function updateWyrPrompt(id: string, data: { optionA?: string; optionB?: string }): Promise<WYRPrompt> {
  const response = await apiFetch<{ prompt: WYRPrompt }>(`/wyr/${id}`, {
    method: 'PATCH',
    body: JSON.stringify(data),
  });
  if (!response.success) {
    throw new Error(response.error?.message || 'Failed to update WYR prompt');
  }
  return response.data.prompt;
}

/**
 * Delete WYR prompt (admin only)
 */
export async function deleteWyrPrompt(id: string): Promise<void> {
  const response = await apiFetch<Record<string, never>>(`/wyr/${id}`, {
    method: 'DELETE',
  });
  if (!response.success) {
    throw new Error(response.error?.message || 'Failed to delete WYR prompt');
  }
}
```

Create comprehensive WYR state management hook:

**useWouldYouRather.ts:**

Props: { envelopeId: string }

Returns:
- state: WYRState (prompt, phase, myVote, partnerVoted, results)
- isLoading: boolean
- error: string | null
- isConnected: boolean (from SignalR)
- vote: (choice: WYRChoice) => Promise<void>
- advance: () => void (for completing after reveal)

Implementation:

1. **Initial load:**
   - Fetch prompt state via api.getWyrPrompt(envelopeId)
   - Determine initial phase:
     - No myVote -> 'voting'
     - myVote but !results -> 'waiting'
     - results -> 'revealing' or 'complete'

2. **SignalR subscriptions (via useSignalREvent):**
   - 'wyrVoteSubmitted': When partner votes, set partnerVoted = true
   - 'wyrRevealReady': Receive results, transition to 'revealing'

3. **Vote action:**
   - Validate isConnected
   - Call api.submitWyrVote(promptId, choice)
   - On success:
     - Set myVote = choice
     - If response.revealed: set results and phase = 'revealing'
     - Else: set phase = 'waiting'
   - On error: set error state

4. **Advance action:**
   - Set phase = 'complete'
   - Trigger envelope refresh (parent will see completed status)

5. **Connection handling:**
   - Get isConnected from useSignalRConnection
   - Track connectionState for reconnecting indicator

6. **First-time hint:**
   - Track whether user has voted before (localStorage key: 'wyr_hint_shown')
   - Return showHint boolean for VotingPhase

**strings.ts additions:**
- WYR_ERROR_LOADING: 'Failed to load activity'
- WYR_ERROR_VOTING: 'Failed to submit vote'
- WYR_ERROR_OFFLINE: 'You need to be connected to vote'
